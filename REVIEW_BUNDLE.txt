     3	import java.io.ByteArrayOutputStream;
     4	import java.util.ArrayList;
     5	import java.util.List;
     6	
     7	/** H.264 Annex-B (start-code) helpers. */
     8	final class AnnexB {
     9	    private AnnexB() {}
    10	
    11	    static int type(byte[] nal) { return nal.length > 0 ? nal[0] & 0x1F : -1; }
    12	
    13	    /** Splits an Annex-B buffer into NAL units without start codes. */
    14	    static List<byte[]> split(byte[] b, int off, int len) {
    15	        List<byte[]> out = new ArrayList<>();
    16	        int end = off + len;
    17	        int nalStart = -1;
    18	        int i = off;
    19	        while (i + 2 < end) {
    20	            if (b[i] == 0 && b[i + 1] == 0 && b[i + 2] == 1) {
    21	                if (nalStart >= 0) add(out, b, nalStart, i);
    22	                i += 3;
    23	                nalStart = i;
    24	            } else {
    25	                i++;
    26	            }
    27	        }
    28	        if (nalStart >= 0) add(out, b, nalStart, end);
    29	        else if (len > 0) add(out, b, off, end); // no start code: treat as one NAL
    30	        return out;
    31	    }
    32	
    33	    private static void add(List<byte[]> out, byte[] b, int s, int e) {
    34	        while (e > s && b[e - 1] == 0) e--; // trailing zeros / leading zero of 4-byte start code
    35	        if (e > s) {
    36	            byte[] n = new byte[e - s];
    37	            System.arraycopy(b, s, n, 0, n.length);
    38	            out.add(n);
    39	        }
    40	    }
    41	
    42	    /** 4-byte length-prefixed sample; drops SPS(7), PPS(8), AUD(9). */
    43	    static byte[] toAvcc(List<byte[]> nals) {
    44	        ByteArrayOutputStream o = new ByteArrayOutputStream();
    45	        for (byte[] n : nals) {
    46	            int t = type(n);
    47	            if (t == 7 || t == 8 || t == 9) continue;
    48	            int l = n.length;
    49	            o.write(l >>> 24); o.write(l >>> 16); o.write(l >>> 8); o.write(l);
    50	            o.write(n, 0, l);
    51	        }
    52	        return o.toByteArray();
    53	    }
    54	}

   116	        AudioTimestamp ts = new AudioTimestamp();

   118	        // Capture time of a frame from AudioRecord's own timestamp (fallback: estimate from backlog).

   122	                if (Math.abs(t - nowNs) < 1_000_000_000L) return t;
   123	            }

   142	        int idx = -1;

   171	                        }
   172	                        inited = true;
   173	                    }
   174	                    continue;
   175	                }
   176	                if (idx < 0) continue;
   177	                ByteBuffer bb = codec.getOutputBuffer(idx);
   178	                byte[] data = new byte[info.size];
   179	                if (bb != null && info.size > 0) {
   180	                    bb.position(info.offset);
   181	                    bb.limit(info.offset + info.size);
   182	                    bb.get(data);
   183	                }
   184	                int flags = info.flags;
   185	                long pts = info.presentationTimeUs;
   186	                codec.releaseOutputBuffer(idx, false);
   187	                if ((flags & MediaCodec.BUFFER_FLAG_END_OF_STREAM) != 0) break;
   188	                if (data.length == 0) continue;
   189	                if ((flags & MediaCodec.BUFFER_FLAG_CODEC_CONFIG) != 0) {
   190	                    if (!java.util.Arrays.equals(data, lastAsc)) {   // avoid a second identical init
   191	                        sink.onAudioInit(Fmp4.audioInit(data, RATE, CHANNELS), "mp4a.40.2");
   192	                        lastAsc = data;
   193	                    }
   194	                    inited = true;
   195	                    continue;
   196	                }
   197	                if (!inited) continue;

   200	                sink.onAudioFragment(Fmp4.fragment(seq++, next, 1024, data, Fmp4.FLAGS_SYNC));
   201	                next += 1024;
   202	            }
   203	        } catch (Exception e) {

   205	        }

===== FILE: app/src/main/java/com/apptesla/mirror/AudioPacer.java =====
     1	package com.apptesla.mirror;
     2	
     3	/**
     4	 * Turns irregular PCM delivery into a gap-free sequence of 1024-sample chunks with
     5	 * capture-time timestamps (System.nanoTime base).
     6	 *  - Normal jitter: chunks are emitted only when full, so no silence is ever inserted
     7	 *    into continuous audio.
     8	 *  - Real silence (no data arriving): silent chunks keep the timeline near real time,
     9	 *    so the browser's audio track never stalls.
    10	 *  - Data older than the timeline (late burst after silence fill) is dropped; a timeline
    11	 *    behind the capture clock is filled with silence. This bounds latency and drift.
    12	 * Pure Java (no Android APIs) so it can be unit-tested off-device.
    13	 */
    14	final class AudioPacer {
    15	    interface Clock { long captureNs(long frameIndex, long nowNs); }
    16	    interface Out { void emit(byte[] pcm, int off, long ptsNs, boolean silence); }
    17	
    18	    static final int RATE = 48000, FRAME_BYTES = 4;              // 16-bit stereo
    19	    static final int CHUNK = 1024 * FRAME_BYTES;
    20	    static final long CHUNK_NS = 1024L * 1_000_000_000L / RATE;  // 21.333 ms
    21	    static final long RESYNC_NS = 60_000_000L;
    22	    static final long SILENCE_AFTER_NS = 150_000_000L;
    23	    static final long MAX_FILL_NS = 1_000_000_000L;
    24	
    25	    private final byte[] acc = new byte[CHUNK * 16];
    26	    private final byte[] silence = new byte[CHUNK];
    27	    private int accLen;
    28	    private long readFrames;
    29	    private long tNs = Long.MIN_VALUE;
    30	    private final Clock clock;
    31	    private final Out out;
    32	
    33	    AudioPacer(Clock clock, Out out) { this.clock = clock; this.out = out; }
    34	
    35	    long readFrames() { return readFrames; }
    36	    int backlogFrames() { return accLen / FRAME_BYTES; }
    37	    int free() { return acc.length - accLen; }
    38	
    39	    void append(byte[] b, int off, int n) {
    40	        n -= n % FRAME_BYTES;
    41	        if (n <= 0) return;
    42	        if (n > acc.length) { off += n - acc.length; readFrames += (n - acc.length) / FRAME_BYTES; n = acc.length; }
    43	        if (accLen + n > acc.length) drop(accLen + n - acc.length);
    44	        System.arraycopy(b, off, acc, accLen, n);
    45	        accLen += n;
    46	        readFrames += n / FRAME_BYTES;
    47	    }
    48	
    49	    void process(long nowNs) {
    50	        while (true) {
    51	            if (accLen >= CHUNK) {
    52	                long head = readFrames - accLen / FRAME_BYTES;
    53	                long cap = clock.captureNs(head, nowNs);
    54	                if (tNs == Long.MIN_VALUE) tNs = cap;
    55	                long d = cap - tNs;
    56	                if (d > RESYNC_NS) {
    57	                    if (d > MAX_FILL_NS) tNs = cap;
    58	                    else { emit(silence, 0, true); continue; }
    59	                } else if (d < -RESYNC_NS) {
    60	                    drop(CHUNK);
    61	                    continue;
    62	                }
    63	                emit(acc, 0, false);
    64	                drop(CHUNK);
    65	            } else {
    66	                if (tNs != Long.MIN_VALUE && nowNs - tNs > SILENCE_AFTER_NS) { emit(silence, 0, true); continue; }
    67	                return;
    68	            }
    69	        }
    70	    }
    71	
    72	    private void emit(byte[] b, int off, boolean isSilence) {
    73	        out.emit(b, off, tNs, isSilence);
    74	        tNs += CHUNK_NS;
    75	    }
    76	
    77	    private void drop(int bytes) {
    78	        bytes = Math.min(bytes, accLen);
    79	        System.arraycopy(acc, bytes, acc, 0, accLen - bytes);
    80	        accLen -= bytes;
    81	    }
    82	}

    24	        W u24(int v) { u8(v >>> 16); u8(v >>> 8); u8(v); return this; }

    26	        W u64(long v) { u32(v >>> 32); u32(v & 0xFFFFFFFFL); return this; }

    28	        W str(String s) { byte[] b = s.getBytes(StandardCharsets.US_ASCII); o.write(b, 0, b.length); return this; }
    29	        W bytes(byte[] b) { o.write(b, 0, b.length); return this; }

    34	    static byte[] concat(byte[]... parts) {
    35	        int n = 0;
    36	        for (byte[] p : parts) n += p.length;
    37	        byte[] r = new byte[n];
    38	        int off = 0;
    39	        for (byte[] p : parts) { System.arraycopy(p, 0, r, off, p.length); off += p.length; }

   210	            for (int i = 1; i < lines.length; i++) {
   211	                int c = lines[i].indexOf(':');
   212	                if (c > 0) hdr.put(lines[i].substring(0, c).trim().toLowerCase(Locale.ROOT), lines[i].substring(c + 1).trim());
   213	            }

   216	                respond(out, "403 Forbidden", "text/plain", "Forbidden".getBytes(StandardCharsets.US_ASCII));
   217	                s.close();
   218	                return;
   219	            }
   220	            if (path.startsWith("/ws") && "websocket".equalsIgnoreCase(hdr.get("upgrade")) && hdr.containsKey("sec-websocket-key")) {
   221	                String origin = hdr.get("origin");
   222	                if (origin != null && !origin.equals("http://" + host)) { // other web pages may not connect
   223	                    respond(out, "403 Forbidden", "text/plain", "Forbidden".getBytes(StandardCharsets.US_ASCII));
   224	                    s.close();
   225	                    return;
   226	                }
   227	                String accept = Base64.getEncoder().encodeToString(MessageDigest.getInstance("SHA-1")
   228	                        .digest((hdr.get("sec-websocket-key") + "258EAFA5-E914-47DA-95CA-C5AB0DC85B11")
   229	                                .getBytes(StandardCharsets.US_ASCII)));
   230	                out.write(("HTTP/1.1 101 Switching Protocols\r\nUpgrade: websocket\r\nConnection: Upgrade\r\n"
   231	                        + "Sec-WebSocket-Accept: " + accept + "\r\n\r\n").getBytes(StandardCharsets.US_ASCII));
   232	                out.flush();

   234	                Client c = new Client(s, in, out);
   235	                c.startThreads();
   236	                join(c);
   237	                return;
   238	            }
   239	            if (path.equals("/") || path.startsWith("/?") || path.startsWith("/index.html")) {
   240	                respond(out, "200 OK", "text/html; charset=utf-8", page);
   241	            } else {
   242	                respond(out, "404 Not Found", "text/plain", "Not found".getBytes(StandardCharsets.US_ASCII));
   243	            }
   244	            s.close();
   245	        } catch (Exception e) {
   246	            try { s.close(); } catch (IOException ignored) {}
   247	        }

   250	    /** Host header must be a literal IP (optionally with port), never a domain name. */
   251	    static boolean isIpHost(String host) {
   252	        if (host == null) return false;
   253	        String h = host.trim();
   254	        if (h.startsWith("[")) {                       // [IPv6]:port
   255	            int e = h.indexOf(']');
   256	            if (e < 0) return false;
   257	            String rest = h.substring(e + 1);
   258	            if (!rest.isEmpty() && !rest.matches(":\\d{1,5}")) return false;
   259	            return h.substring(1, e).matches("[0-9a-fA-F:.%a-zA-Z0-9]+") && h.substring(1, e).contains(":");
   260	        }
   261	        return h.matches("(25[0-5]|2[0-4]\\d|1?\\d?\\d)(\\.(25[0-5]|2[0-4]\\d|1?\\d?\\d)){3}(:\\d{1,5})?");
   262	    }

   266	        if (a == null) return false;
   267	        if (a.isLoopbackAddress() || a.isSiteLocalAddress() || a.isLinkLocalAddress()) return true;

   274	                + "\r\nCache-Control: no-store\r\nConnection: close\r\n\r\n").getBytes(StandardCharsets.US_ASCII));

   279	    private static String readHead(InputStream in) throws IOException {
   280	        ByteArrayOutputStream b = new ByteArrayOutputStream();
   281	        int state = 0;
   282	        while (b.size() < 16384) {
   283	            int c = in.read();
   284	            if (c < 0) return null;
   285	            b.write(c);
   286	            state = (c == '\r' && (state == 0 || state == 2)) ? state + 1
   287	                  : (c == '\n' && (state == 1 || state == 3)) ? state + 1 : 0;
   288	            if (state == 4) return b.toString("ISO-8859-1");
   289	        }
   290	        return null;
   291	    }

   293	    private final class Client {
   294	        final Socket s; final InputStream in; final OutputStream out;
   295	        final LinkedBlockingQueue<byte[]> queue = new LinkedBlockingQueue<>();
   296	        volatile boolean waitKey = true, configured = false, open = true;
   297	        long lastRecoverMs;
   298	
   299	        Client(Socket s, InputStream in, OutputStream out) { this.s = s; this.in = in; this.out = out; }
   300	
   301	        boolean offer(byte[] f) {
   302	            if (queue.size() >= QUEUE_LIMIT) return false;
   303	            return queue.offer(f);
   304	        }
   305	
   306	        void startThreads() {
   307	            Thread w = new Thread(() -> {
   308	                try {
   309	                    long lastPing = System.currentTimeMillis();
   310	                    while (open) {
   311	                        byte[] f = queue.poll(500, TimeUnit.MILLISECONDS);
   312	                        if (f != null) out.write(f);
   313	                        long now = System.currentTimeMillis();
   314	                        if (now - lastPing >= 5000) { out.write(PING); lastPing = now; }
   315	                    }
   316	                } catch (Exception e) { /* fall through */ }
   317	                close();
   318	            }, "ws-writer");
   319	            w.setDaemon(true);
   320	            w.start();
   321	            Thread r = new Thread(() -> {
   322	                try { readFrames(); } catch (Exception ignored) {}
   323	                close();
   324	            }, "ws-reader");
   325	            r.setDaemon(true);
   326	            r.start();
   327	        }
   328	
   329	        /** Reads and discards client frames; returns on close/EOF. */
   330	        private void readFrames() throws IOException {
   331	            while (open) {
   332	                int b0 = in.read(), b1 = in.read();
   333	                if (b0 < 0 || b1 < 0) return;
   334	                long len = b1 & 0x7F;
   335	                if (len == 126) len = (rd(in) << 8) | rd(in);
   336	                else if (len == 127) { len = 0; for (int i = 0; i < 8; i++) len = (len << 8) | rd(in); }
   337	                if ((b1 & 0x80) != 0) for (int i = 0; i < 4; i++) rd(in);
   338	                for (long i = 0; i < len; i++) rd(in);
   339	                if ((b0 & 0x0F) == 0x8) return; // close
   340	            }
   341	        }
   342	
   343	        private int rd(InputStream in) throws IOException {
   344	            int c = in.read();
   345	            if (c < 0) throw new IOException("eof");
   346	            return c;
   347	        }
   348	
   349	        void close() {
   350	            if (!open) return;
   351	            open = false;
   352	            clients.remove(this);
   353	            try { s.close(); } catch (IOException ignored) {}
   354	        }
   355	    }

    33	    private VideoEncoder(MediaCodec codec, int w, int h, long t0Us, Sink sink) {
    34	        this.codec = codec; this.width = w; this.height = h; this.t0Us = t0Us; this.sink = sink;
    35	        this.surface = codec.createInputSurface();

    38	    /** Picks the largest size (long side <= 1920) the hardware encoder accepts. */
    39	    static VideoEncoder create(int srcW, int srcH, long t0Us, Sink sink) throws IOException {
    40	        MediaCodec codec = MediaCodec.createEncoderByType(MIME);
    41	        MediaCodecInfo.VideoCapabilities caps =
    42	                codec.getCodecInfo().getCapabilitiesForType(MIME).getVideoCapabilities();
    43	        for (int max : MAX_SIDES) {
    44	            double scale = Math.min(1.0, (double) max / Math.max(srcW, srcH));
    45	            int w = ((int) (srcW * scale)) & ~15;
    46	            int h = ((int) (srcH * scale)) & ~15;
    47	            if (w < 160 || h < 160) continue;
    48	            if (caps != null && !caps.areSizeAndRateSupported(w, h, 30)) continue;
    49	            boolean ok = false;
    50	            for (boolean repeat : new boolean[]{true, false}) { // some encoders reject REPEAT_PREVIOUS_FRAME
    51	                try {
    52	                    codec.configure(format(w, h, repeat), null, null, MediaCodec.CONFIGURE_FLAG_ENCODE);
    53	                    ok = true;
    54	                    break;
    55	                } catch (Exception e) {
    56	                    Log.w(TAG, "configure failed for " + w + "x" + h + " repeat=" + repeat, e);
    57	                    codec.reset();
    58	                }
    59	            }
    60	            if (!ok) continue;
    61	            try {
    62	                Log.i(TAG, "video encoder " + codec.getName() + " " + w + "x" + h);
    63	                return new VideoEncoder(codec, w, h, t0Us, sink);
    64	            } catch (Exception e) {          // createInputSurface failed
    65	                codec.release();
    66	                throw new IOException("Video kodlayıcı yüzeyi oluşturulamadı", e);
    67	            }
    68	        }
    69	        codec.release();
    70	        throw new IOException("Uygun video kodlayıcı boyutu bulunamadı");
    71	    }

   111	    private void drain() {
   112	        MediaCodec.BufferInfo info = new MediaCodec.BufferInfo();
   113	        byte[] sps = null, pps = null;
   114	        boolean inited = false;
   115	        long offsetUs = Long.MIN_VALUE;
   116	        int seq = 1;
   117	        long pendT = -1, lastEnd = 0, pendArrivalNs = 0; byte[] pendSample = null; boolean pendKey = false;
   118	        try {
   119	            while (!stop) {
   120	                int idx = codec.dequeueOutputBuffer(info, 10_000);
   121	                if (idx == MediaCodec.INFO_OUTPUT_FORMAT_CHANGED) {
   122	                    // Some encoders deliver SPS/PPS only as csd-0/csd-1 in the output format.
   123	                    MediaFormat of = codec.getOutputFormat();
   124	                    for (String k : new String[]{"csd-0", "csd-1"}) {
   125	                        ByteBuffer c = of.containsKey(k) ? of.getByteBuffer(k) : null;
   126	                        if (c == null) continue;
   127	                        byte[] b = new byte[c.remaining()];
   128	                        c.duplicate().get(b);
   129	                        for (byte[] n : AnnexB.split(b, 0, b.length)) {
   130	                            if (AnnexB.type(n) == 7 && sps == null) sps = n;
   131	                            else if (AnnexB.type(n) == 8 && pps == null) pps = n;
   132	                        }
   133	                    }
   134	                    if (!inited && sps != null && pps != null) {
   135	                        sink.onVideoInit(Fmp4.videoInit(sps, pps, width, height), Fmp4.videoCodec(sps));
   136	                        inited = true;
   137	                    }
   138	                    continue;
   139	                }
   140	                if (idx < 0) {
   141	                    // Screen went still: don't hold the last frame back, publish it now.
   142	                    if (pendSample != null && System.nanoTime() - pendArrivalNs > 50_000_000L) {
   143	                        long nowT = (System.nanoTime() / 1000 - t0Us) * 9 / 100;
   144	                        int dur = (int) Math.max(3000, Math.min(nowT - pendT, 90_000));
   145	                        sink.onVideoFragment(Fmp4.fragment(seq++, pendT, dur, pendSample,
   146	                                pendKey ? Fmp4.FLAGS_SYNC : Fmp4.FLAGS_NON_SYNC), pendKey);
   147	                        lastEnd = pendT + dur;
   148	                        pendSample = null;
   149	                    }
   150	                    continue;
   151	                }
   152	                ByteBuffer bb = codec.getOutputBuffer(idx);
   153	                byte[] data = new byte[info.size];
   154	                if (bb != null && info.size > 0) {
   155	                    bb.position(info.offset);
   156	                    bb.limit(info.offset + info.size);
   157	                    bb.get(data);
   158	                }
   159	                int flags = info.flags;
   160	                long pts = info.presentationTimeUs;
   161	                codec.releaseOutputBuffer(idx, false);
   162	                if ((flags & MediaCodec.BUFFER_FLAG_END_OF_STREAM) != 0) break;
   163	                if (data.length == 0) continue;
   164	
   165	                List<byte[]> nals = AnnexB.split(data, 0, data.length);
   166	                boolean key = (flags & MediaCodec.BUFFER_FLAG_KEY_FRAME) != 0;
   167	                for (byte[] n : nals) {
   168	                    int t = AnnexB.type(n);
   169	                    if (t == 7) sps = n;
   170	                    else if (t == 8) pps = n;
   171	                    else if (t == 5) key = true;
   172	                }
   173	                if (!inited && sps != null && pps != null) {
   174	                    sink.onVideoInit(Fmp4.videoInit(sps, pps, width, height), Fmp4.videoCodec(sps));
   175	                    inited = true;
   176	                }
   177	                if ((flags & MediaCodec.BUFFER_FLAG_CODEC_CONFIG) != 0 || !inited) continue;
   178	
   179	                byte[] sample = AnnexB.toAvcc(nals);
   180	                if (sample.length == 0) continue;
   181	
   182	                if (offsetUs == Long.MIN_VALUE) {
   183	                    long now = System.nanoTime() / 1000;
   184	                    offsetUs = Math.abs(now - pts) > 1_000_000 ? now - pts : 0;
   185	                    if (offsetUs != 0) Log.w(TAG, "video pts not on monotonic clock, offset " + offsetUs);
   186	                }
   187	                long t = Math.max(0, (pts + offsetUs - t0Us) * 9 / 100); // us -> 90 kHz
   188	                if (pendSample == null && t <= lastEnd - 3000) t = lastEnd; // never go backwards after an idle flush
   189	
   190	                if (pendSample != null) {
   191	                    if (t <= pendT) t = pendT + 1;
   192	                    int dur = (int) Math.min(t - pendT, 90_000);
   193	                    sink.onVideoFragment(Fmp4.fragment(seq++, pendT, dur, pendSample,
   194	                            pendKey ? Fmp4.FLAGS_SYNC : Fmp4.FLAGS_NON_SYNC), pendKey);
   195	                }
   196	                pendT = t; pendSample = sample; pendKey = key; pendArrivalNs = System.nanoTime();
   197	            }
   198	        } catch (Exception e) {
   199	            if (!stop) Log.e(TAG, "video drain error", e);
   200	        }
   201	    }
   202	}
