package com.apptesla.mirror;
import android.annotation.SuppressLint;
import android.media.AudioAttributes;
import android.media.AudioFormat;
import android.media.AudioPlaybackCaptureConfiguration;
import android.media.AudioRecord;
import android.media.AudioTimestamp;
import android.media.MediaCodec;
import android.media.MediaCodecInfo;
import android.media.MediaFormat;
import android.media.projection.MediaProjection;
import android.util.Log;
import java.nio.ByteBuffer;
/**
 * Internal audio (AudioPlaybackCapture, Android 10+) -> AAC-LC -> fMP4 fragments.
 * AudioPacer turns irregular delivery into gap-free, capture-timestamped chunks.
 */
final class AudioEncoder {
	interface Sink {
		void onAudioInit(byte[] init, String codec);
		void onAudioFragment(byte[] fragment);
	}
	private static final String TAG = "TeslaMirror";
	static final int RATE = 48000, CHANNELS = 2;
	private static final int FRAME_BYTES = CHANNELS * 2;          // 16-bit stereo
	private static final int CHUNK = AudioPacer.CHUNK;            // 1024 samples ≈ 21.3 ms
	private final AudioRecord rec;
	private final MediaCodec codec;
	private final long t0Us;
	private final Sink sink;
	private volatile boolean stop;
	private Thread inThread, outThread;
	@SuppressLint("MissingPermission") // RECORD_AUDIO checked by caller
	AudioEncoder(MediaProjection projection, long t0Us, Sink sink) throws Exception {
		this.t0Us = t0Us;
		this.sink = sink;
		AudioPlaybackCaptureConfiguration cfg = new AudioPlaybackCaptureConfiguration.Builder(projection)
		        .addMatchingUsage(AudioAttributes.USAGE_MEDIA)
		        .addMatchingUsage(AudioAttributes.USAGE_GAME)
		        .addMatchingUsage(AudioAttributes.USAGE_UNKNOWN)
		        .build();
		AudioFormat fmt = new AudioFormat.Builder()
		        .setEncoding(AudioFormat.ENCODING_PCM_16BIT)
		        .setSampleRate(RATE)
		        .setChannelMask(AudioFormat.CHANNEL_IN_STEREO)
		        .build();
		int min = AudioRecord.getMinBufferSize(RATE, AudioFormat.CHANNEL_IN_STEREO, AudioFormat.ENCODING_PCM_16BIT);
		rec = new AudioRecord.Builder()
		        .setAudioFormat(fmt)
		        .setBufferSizeInBytes(Math.max(min * 2, CHUNK * 8))
		        .setAudioPlaybackCaptureConfig(cfg)
		        .build();
		if (rec.getState() != AudioRecord.STATE_INITIALIZED) {
		    rec.release();
		    throw new IllegalStateException("AudioRecord başlatılamadı");
		}
		MediaFormat f = MediaFormat.createAudioFormat(MediaFormat.MIMETYPE_AUDIO_AAC, RATE, CHANNELS);
		f.setInteger(MediaFormat.KEY_AAC_PROFILE, MediaCodecInfo.CodecProfileLevel.AACObjectLC);
		f.setInteger(MediaFormat.KEY_BIT_RATE, 160_000);
		f.setInteger(MediaFormat.KEY_MAX_INPUT_SIZE, CHUNK * 2);
		MediaCodec c;
		try {
		    c = MediaCodec.createEncoderByType(MediaFormat.MIMETYPE_AUDIO_AAC);
		} catch (Exception e) {
		    rec.release();
		    throw e;
		}
		codec = c;
		try {
		    codec.configure(f, null, null, MediaCodec.CONFIGURE_FLAG_ENCODE);
		} catch (Exception e) {
		    codec.release();
		    rec.release();
		    throw e;
		}
	}
	/** On failure everything is released and the exception is rethrown (caller drops audio). */
	void start() {
		try {
		    rec.startRecording();
		    if (rec.getRecordingState() != AudioRecord.RECORDSTATE_RECORDING)
		        throw new IllegalStateException("AudioRecord kayda başlamadı");
		    codec.start();
		    inThread = new Thread(this::feed, "audio-in");
		    outThread = new Thread(this::drain, "audio-out");
		    inThread.start();
		    outThread.start();
		} catch (RuntimeException e) {
		    stop();
		    throw e;
		}
	}
	private boolean released;
	synchronized void stop() {
		if (released) return;
		released = true;
		stop = true;
		try { if (inThread != null) inThread.join(1000); } catch (InterruptedException ignored) {}
		try { if (outThread != null) outThread.join(1000); } catch (InterruptedException ignored) {}
		try { rec.stop(); } catch (Exception ignored) {}
		rec.release();
		try { codec.stop(); } catch (Exception ignored) {}
		try { codec.release(); } catch (Exception ignored) {}
	}
	private void feed() {
		AudioTimestamp ts = new AudioTimestamp();
		final AudioPacer[] self = new AudioPacer[1];
		// Capture time of a frame from AudioRecord's own timestamp (fallback: estimate from backlog).
		AudioPacer.Clock clock = (frame, nowNs) -> {
		    if (rec.getTimestamp(ts, AudioTimestamp.TIMEBASE_MONOTONIC) == AudioRecord.SUCCESS) {
		        long t = ts.nanoTime + (frame - ts.framePosition) * 1_000_000_000L / RATE;
		        if (Math.abs(t - nowNs) < 1_000_000_000L) return t;
		    }
		    return nowNs - (self[0].readFrames() - frame) * 1_000_000_000L / RATE;
		};
		AudioPacer pacer = new AudioPacer(clock, (pcm, off, ptsNs, silent) -> queue(pcm, off, ptsNs / 1000));
		self[0] = pacer;
		byte[] buf = new byte[CHUNK * 4];
		try {
		    while (!stop) {
		        int n = rec.read(buf, 0, Math.min(buf.length, pacer.free()), AudioRecord.READ_NON_BLOCKING);
		        if (n > 0) pacer.append(buf, 0, n);
		        pacer.process(System.nanoTime());
		        if (n <= 0) Thread.sleep(5);
		    }
		} catch (Exception e) {
		    if (!stop) Log.e(TAG, "audio feed error", e);
		}
	}
	private void queue(byte[] pcm, int off, long ptsUs) {
		int idx = -1;
		while (!stop && idx < 0) idx = codec.dequeueInputBuffer(20_000);
		if (idx < 0) return;
		ByteBuffer ib = codec.getInputBuffer(idx);
		if (ib == null) return;
		ib.clear();
		ib.put(pcm, off, CHUNK);
		codec.queueInputBuffer(idx, 0, CHUNK, ptsUs, 0);
	}
	private void drain() {
		MediaCodec.BufferInfo info = new MediaCodec.BufferInfo();
		boolean inited = false;
		byte[] lastAsc = null;
		long next = -1; // in 48 kHz units
		int seq = 1;
		try {
		    while (!stop) {
		        int idx = codec.dequeueOutputBuffer(info, 10_000);
		        if (idx == MediaCodec.INFO_OUTPUT_FORMAT_CHANGED) {
		            // Some encoders deliver the AudioSpecificConfig only as csd-0 in the output format.
		            MediaFormat of = codec.getOutputFormat();
		            ByteBuffer c = of.containsKey("csd-0") ? of.getByteBuffer("csd-0") : null;
		            if (c != null && c.remaining() > 0) {
		                byte[] asc = new byte[c.remaining()];
		                c.duplicate().get(asc);
		                if (!java.util.Arrays.equals(asc, lastAsc)) {
		                    sink.onAudioInit(Fmp4.audioInit(asc, RATE, CHANNELS), "mp4a.40.2");
		                    lastAsc = asc;
		                }
		                inited = true;
		            }
		            continue;
		        }
		        if (idx < 0) continue;
		        ByteBuffer bb = codec.getOutputBuffer(idx);
		        byte[] data = new byte[info.size];
		        if (bb != null && info.size > 0) {
		            bb.position(info.offset);
		            bb.limit(info.offset + info.size);
		            bb.get(data);
		        }
		        int flags = info.flags;
		        long pts = info.presentationTimeUs;
		        codec.releaseOutputBuffer(idx, false);
		        if ((flags & MediaCodec.BUFFER_FLAG_END_OF_STREAM) != 0) break;
		        if (data.length == 0) continue;
		        if ((flags & MediaCodec.BUFFER_FLAG_CODEC_CONFIG) != 0) {
		            if (!java.util.Arrays.equals(data, lastAsc)) {   // avoid a second identical init
		                sink.onAudioInit(Fmp4.audioInit(data, RATE, CHANNELS), "mp4a.40.2");
		                lastAsc = data;
		            }
		            inited = true;
		            continue;
		        }
		        if (!inited) continue;
		        long t = Math.max(0, (pts - t0Us) * RATE / 1_000_000);
		        if (next < 0 || Math.abs(t - next) > RATE / 10) next = t; // resync if >100 ms off
		        sink.onAudioFragment(Fmp4.fragment(seq++, next, 1024, data, Fmp4.FLAGS_SYNC));
		        next += 1024;
		    }
		} catch (Exception e) {
		    if (!stop) Log.e(TAG, "audio drain error", e);
		}
	}
}

/**
 * Tiny dependency-free HTTP + WebSocket server.
 * GET /    -> player page
 * GET /ws  -> WebSocket; text = JSON config, binary = [type byte][payload]
 *            type 1 video init, 2 video fragment, 3 audio init, 4 audio fragment
 */
final class StreamServer {
	interface KeyFrameRequester { void requestKeyFrame(); }
	static final byte T_VIDEO_INIT = 1, T_VIDEO = 2, T_AUDIO_INIT = 3, T_AUDIO = 4;
	private static final int QUEUE_LIMIT = 240;
	private static final byte[] PING = frame(0x9, new byte[0]);
	private final int port;
	private final byte[] page;
	private final KeyFrameRequester keyReq;
	private final CopyOnWriteArrayList<Client> clients = new CopyOnWriteArrayList<>();
	private volatile ServerSocket server;
	private volatile boolean running;
	// Current stream state (guarded by this)
	private String videoCodec, audioCodec;
	private byte[] videoInitFrame, audioInitFrame;
	private boolean audioExpected = true;
	private volatile String lang = "en";
	/** Language of the Tesla page; pushed live to connected pages. */
	synchronized void setLang(String code) {
		if (code == null || !code.matches("[a-z]{2}")) return;
		lang = code;
		byte[] f = frame(0x1, ("{\"type\":\"lang\",\"lang\":\"" + code + "\"}").getBytes(StandardCharsets.UTF_8));
		for (Client c : clients) c.queue.offer(f);
	}
	StreamServer(int port, byte[] page, KeyFrameRequester keyReq) {
		this.port = port;
		this.page = page;
		this.keyReq = keyReq;
	}
	void start() throws IOException {
		ServerSocket ss = new ServerSocket();
		ss.setReuseAddress(true);
		IOException last = null;
		for (int i = 0; i < 10; i++) {          // previous session may still be releasing the port
		    try { ss.bind(new InetSocketAddress(port)); last = null; break; }
		    catch (IOException e) { last = e; try { Thread.sleep(300); } catch (InterruptedException ignored) {} }
		}
		if (last != null) { ss.close(); throw last; }
		server = ss;
		running = true;
		Thread t = new Thread(this::acceptLoop, "ws-accept");
		t.setDaemon(true);
		t.start();
	}
	void stop() {
		running = false;
		try { if (server != null) server.close(); } catch (IOException ignored) {}
		for (Client c : clients) c.close();
		clients.clear();
	}
	int clientCount() { return clients.size(); }
	// ---------------- stream state ----------------
	/** Call with false if audio capture is unavailable, so clients don't wait for it. */
	synchronized void setAudioExpected(boolean expected) {
		audioExpected = expected;
		if (!expected) { audioCodec = null; audioInitFrame = null; }
		broadcastConfig();
	}
	synchronized void setVideoInit(byte[] init, String codec) {
		videoCodec = codec;
		videoInitFrame = frame(0x2, typed(T_VIDEO_INIT, init));
		broadcastConfig();
	}
	synchronized void setAudioInit(byte[] init, String codec) {
		audioCodec = codec;
		audioInitFrame = frame(0x2, typed(T_AUDIO_INIT, init));
		broadcastConfig();
	}
	synchronized void sendVideo(byte[] fragment, boolean key) {
		if (clients.isEmpty() || videoInitFrame == null) return;
		byte[] f = frame(0x2, typed(T_VIDEO, fragment));
		for (Client c : clients) {
		    if (!c.configured) continue;
		    if (c.waitKey) { if (!key) continue; c.waitKey = false; }
		    if (!c.offer(f)) recover(c);
		}
	}
	synchronized void sendAudio(byte[] fragment) {
		if (clients.isEmpty() || audioInitFrame == null) return;
		byte[] f = frame(0x2, typed(T_AUDIO, fragment));
		for (Client c : clients) {
		    if (!c.configured || c.waitKey) continue; // start audio together with first video keyframe
		    if (!c.offer(f)) recover(c);
		}
	}
	private boolean ready() {
		return videoInitFrame != null && (!audioExpected || audioInitFrame != null);
	}
	/** (Re)sends config + init segments; client rebuilds its MediaSource. */
	private void broadcastConfig() {
		if (!ready()) return;
		for (Client c : clients) configure(c);
		keyReq.requestKeyFrame();
	}
	private void configure(Client c) {
		c.queue.clear();
		String json = "{\"type\":\"config\",\"v\":\"" + videoCodec + "\",\"a\":"
		        + (audioInitFrame != null ? "\"" + audioCodec + "\"" : "null") + "}";
		c.queue.offer(frame(0x1, ("{\"type\":\"lang\",\"lang\":\"" + lang + "\"}").getBytes(StandardCharsets.UTF_8)));
		c.queue.offer(frame(0x1, json.getBytes(StandardCharsets.UTF_8)));
		c.queue.offer(videoInitFrame);
		if (audioInitFrame != null) c.queue.offer(audioInitFrame);
		c.waitKey = true;
		c.configured = true;
	}
	/** Slow client: drop backlog, restart from next keyframe. */
	private void recover(Client c) {
		long now = System.currentTimeMillis();
		if (now - c.lastRecoverMs < 5000) { c.close(); return; } // persistently stuck viewer
		c.lastRecoverMs = now;
		configure(c);
		keyReq.requestKeyFrame();
	}
	private synchronized void join(Client c) {
		if (!running || !c.open) { c.close(); return; }
		clients.add(c);
		if (!c.open) { clients.remove(c); return; } // closed while registering
		if (ready()) { configure(c); keyReq.requestKeyFrame(); }
	}
	private static byte[] typed(byte type, byte[] payload) {
		byte[] r = new byte[payload.length + 1];
		r[0] = type;
		System.arraycopy(payload, 0, r, 1, payload.length);
		return r;
	}
	/** Server->client WebSocket frame (unmasked, FIN). */
	static byte[] frame(int opcode, byte[] p) {
		int n = p.length;
		int h = n < 126 ? 2 : (n < 65536 ? 4 : 10);
		byte[] f = new byte[h + n];
		f[0] = (byte) (0x80 | opcode);
		if (n < 126) {
		    f[1] = (byte) n;
		} else if (n < 65536) {
		    f[1] = 126; f[2] = (byte) (n >>> 8); f[3] = (byte) n;
		} else {
		    f[1] = 127;
		    long L = n;
		    for (int i = 0; i < 8; i++) f[2 + i] = (byte) (L >>> (56 - 8 * i));
		}
		System.arraycopy(p, 0, f, h, n);
		return f;
	}
	// ---------------- networking ----------------
	private void acceptLoop() {
		while (running) {
		    try {
		        Socket s = server.accept();
		        Thread t = new Thread(() -> handle(s), "http-conn");
		        t.setDaemon(true);
		        t.start();
		    } catch (IOException e) {
		        if (!running) return;
		    }
		}
	}
	private void handle(Socket s) {
		try {
		    if (!isLocal(s.getInetAddress())) { s.close(); return; } // never serve the internet side
		    s.setTcpNoDelay(true);
		    s.setSoTimeout(15000);
		    InputStream in = s.getInputStream();
		    OutputStream out = s.getOutputStream();
		    String head = readHead(in);
		    if (head == null) { s.close(); return; }
		    String[] lines = head.split("\r\n");
		    String[] req = lines[0].split(" ");
		    String path = req.length > 1 ? req[1] : "/";
		    Map<String, String> hdr = new HashMap<>();
		    for (int i = 1; i < lines.length; i++) {
		        int c = lines[i].indexOf(':');
		        if (c > 0) hdr.put(lines[i].substring(0, c).trim().toLowerCase(Locale.ROOT), lines[i].substring(c + 1).trim());
		    }
		    String host = hdr.get("host");
		    if (!isIpHost(host)) { // DNS-rebinding guard: only http://<IP>:port is served
		        respond(out, "403 Forbidden", "text/plain", "Forbidden".getBytes(StandardCharsets.US_ASCII));
		        s.close();
		        return;
		    }
		    if (path.startsWith("/ws") && "websocket".equalsIgnoreCase(hdr.get("upgrade")) && hdr.containsKey("sec-websocket-key")) {
		        String origin = hdr.get("origin");
		        if (origin != null && !origin.equals("http://" + host)) { // other web pages may not connect
		            respond(out, "403 Forbidden", "text/plain", "Forbidden".getBytes(StandardCharsets.US_ASCII));
		            s.close();
		            return;
		        }
		        String accept = Base64.getEncoder().encodeToString(MessageDigest.getInstance("SHA-1")
		                .digest((hdr.get("sec-websocket-key") + "258EAFA5-E914-47DA-95CA-C5AB0DC85B11")
		                        .getBytes(StandardCharsets.US_ASCII)));
		        out.write(("HTTP/1.1 101 Switching Protocols\r\nUpgrade: websocket\r\nConnection: Upgrade\r\n"
		                + "Sec-WebSocket-Accept: " + accept + "\r\n\r\n").getBytes(StandardCharsets.US_ASCII));
		        out.flush();
		        s.setSoTimeout(15000); // browser answers our pings; silence = dead viewer
		        Client c = new Client(s, in, out);
		        c.startThreads();
		        join(c);
		        return;
		    }
		    if (path.equals("/") || path.startsWith("/?") || path.startsWith("/index.html")) {
		        byte[] body = new String(page, StandardCharsets.UTF_8).replace("__LANG__", lang)
		                .getBytes(StandardCharsets.UTF_8);
		        respond(out, "200 OK", "text/html; charset=utf-8", body);
		    } else {
		        respond(out, "404 Not Found", "text/plain", "Not found".getBytes(StandardCharsets.US_ASCII));
		    }
		    s.close();
		} catch (Exception e) {
		    try { s.close(); } catch (IOException ignored) {}
		}
	}
	/** Host header must be a literal IP (optionally with port), never a domain name. */
	static boolean isIpHost(String host) {
		if (host == null) return false;
		String h = host.trim();
		if (h.startsWith("[")) {                       // [IPv6]:port
		    int e = h.indexOf(']');
		    if (e < 0) return false;
		    String rest = h.substring(e + 1);
		    if (!rest.isEmpty() && !rest.matches(":\\d{1,5}")) return false;
		    return h.substring(1, e).matches("[0-9a-fA-F:.%a-zA-Z0-9]+") && h.substring(1, e).contains(":");
		}
		return h.matches("(25[0-5]|2[0-4]\\d|1?\\d?\\d)(\\.(25[0-5]|2[0-4]\\d|1?\\d?\\d)){3}(:\\d{1,5})?");
	}
	/** Private/link-local/loopback only (hotspot clients), IPv4 and IPv6. */
	static boolean isLocal(InetAddress a) {
		if (a == null) return false;
		if (a.isLoopbackAddress() || a.isSiteLocalAddress() || a.isLinkLocalAddress()) return true;
		if (a instanceof Inet6Address) return (a.getAddress()[0] & 0xFE) == 0xFC; // fc00::/7 ULA
		return false;
	}
	private static void respond(OutputStream out, String status, String type, byte[] body) throws IOException {
		out.write(("HTTP/1.1 " + status + "\r\nContent-Type: " + type + "\r\nContent-Length: " + body.length
		        + "\r\nCache-Control: no-store\r\nConnection: close\r\n\r\n").getBytes(StandardCharsets.US_ASCII));
		out.write(body);
		out.flush();
	}
	private static String readHead(InputStream in) throws IOException {
		ByteArrayOutputStream b = new ByteArrayOutputStream();
		int state = 0;
		while (b.size() < 16384) {
		    int c = in.read();
		    if (c < 0) return null;
		    b.write(c);
		    state = (c == '\r' && (state == 0 || state == 2)) ? state + 1
		          : (c == '\n' && (state == 1 || state == 3)) ? state + 1 : 0;
		    if (state == 4) return b.toString("ISO-8859-1");
		}
		return null;
	}
	private final class Client {
		final Socket s; final InputStream in; final OutputStream out;
		final LinkedBlockingQueue<byte[]> queue = new LinkedBlockingQueue<>();
		volatile boolean waitKey = true, configured = false, open = true;
		long lastRecoverMs;
		Client(Socket s, InputStream in, OutputStream out) { this.s = s; this.in = in; this.out = out; }
		boolean offer(byte[] f) {
		    if (queue.size() >= QUEUE_LIMIT) return false;
		    return queue.offer(f);
		}
		void startThreads() {
		    Thread w = new Thread(() -> {
		        try {
		            long lastPing = System.currentTimeMillis();
		            while (open) {
		                byte[] f = queue.poll(500, TimeUnit.MILLISECONDS);
		                if (f != null) out.write(f);
		                long now = System.currentTimeMillis();
		                if (now - lastPing >= 5000) { out.write(PING); lastPing = now; }
		            }
		        } catch (Exception e) { /* fall through */ }
		        close();
		    }, "ws-writer");
		    w.setDaemon(true);
		    w.start();
		    Thread r = new Thread(() -> {
		        try { readFrames(); } catch (Exception ignored) {}
		        close();
		    }, "ws-reader");
		    r.setDaemon(true);
		    r.start();
		}
		/** Reads and discards client frames; returns on close/EOF. */
		private void readFrames() throws IOException {
		    while (open) {
		        int b0 = in.read(), b1 = in.read();
		        if (b0 < 0 || b1 < 0) return;
		        long len = b1 & 0x7F;
		        if (len == 126) len = (rd(in) << 8) | rd(in);
		        else if (len == 127) { len = 0; for (int i = 0; i < 8; i++) len = (len << 8) | rd(in); }
		        if ((b1 & 0x80) != 0) for (int i = 0; i < 4; i++) rd(in);
		        for (long i = 0; i < len; i++) rd(in);
		        if ((b0 & 0x0F) == 0x8) return; // close
		    }
		}
		private int rd(InputStream in) throws IOException {
		    int c = in.read();
		    if (c < 0) throw new IOException("eof");
		    return c;
		}
		void close() {
		    if (!open) return;
		    open = false;
		    clients.remove(this);
		    try { s.close(); } catch (IOException ignored) {}
		}
	}
}

package com.apptesla.mirror;
import android.media.MediaCodec;
import android.media.MediaCodecInfo;
import android.media.MediaFormat;
import android.os.Bundle;
import android.util.Log;
import android.view.Surface;
import java.io.IOException;
import java.nio.ByteBuffer;
import java.util.List;
/** Screen frames (input Surface) -> H.264 -> fMP4 fragments. */
final class VideoEncoder {
	interface Sink {
		void onVideoInit(byte[] init, String codec);
		void onVideoFragment(byte[] fragment, boolean key);
	}
	private static final String TAG = "TeslaMirror";
	private static final String MIME = MediaFormat.MIMETYPE_VIDEO_AVC;
	private static final int[] MAX_SIDES = {1920, 1600, 1280, 960};
	final int width, height;
	final Surface surface;
	private final MediaCodec codec;
	private final long t0Us;
	private final Sink sink;
	private volatile boolean stop;
	private Thread thread;
	private VideoEncoder(MediaCodec codec, int w, int h, long t0Us, Sink sink) {
		this.codec = codec; this.width = w; this.height = h; this.t0Us = t0Us; this.sink = sink;
		this.surface = codec.createInputSurface();
	}
	/** Picks the largest size (long side <= 1920) the hardware encoder accepts. */
	static VideoEncoder create(int srcW, int srcH, long t0Us, Sink sink) throws IOException {
		MediaCodec codec = MediaCodec.createEncoderByType(MIME);
		MediaCodecInfo.VideoCapabilities caps =
		        codec.getCodecInfo().getCapabilitiesForType(MIME).getVideoCapabilities();
		for (int max : MAX_SIDES) {
		    double scale = Math.min(1.0, (double) max / Math.max(srcW, srcH));
		    int w = ((int) (srcW * scale)) & ~15;
		    int h = ((int) (srcH * scale)) & ~15;
		    if (w < 160 || h < 160) continue;
		    if (caps != null && !caps.areSizeAndRateSupported(w, h, 30)) continue;
		    boolean ok = false;
		    for (boolean repeat : new boolean[]{true, false}) { // some encoders reject REPEAT_PREVIOUS_FRAME
		        try {
		            codec.configure(format(w, h, repeat), null, null, MediaCodec.CONFIGURE_FLAG_ENCODE);
		            ok = true;
		            break;
		        } catch (Exception e) {
		            Log.w(TAG, "configure failed for " + w + "x" + h + " repeat=" + repeat, e);
		            codec.reset();
		        }
		    }
		    if (!ok) continue;
		    try {
		        Log.i(TAG, "video encoder " + codec.getName() + " " + w + "x" + h);
		        return new VideoEncoder(codec, w, h, t0Us, sink);
		    } catch (Exception e) {          // createInputSurface failed
		        codec.release();
		        throw new IOException("Video kodlayıcı yüzeyi oluşturulamadı", e);
		    }
		}
		codec.release();
		throw new IOException("Uygun video kodlayıcı boyutu bulunamadı");
	}
	private static MediaFormat format(int w, int h, boolean repeat) {
		MediaFormat f = MediaFormat.createVideoFormat(MIME, w, h);
		f.setInteger(MediaFormat.KEY_COLOR_FORMAT, MediaCodecInfo.CodecCapabilities.COLOR_FormatSurface);
		int bitrate = (int) Math.max(2_000_000, Math.min(8_000_000, (long) w * h * 30 / 6));
		f.setInteger(MediaFormat.KEY_BIT_RATE, bitrate);
		f.setInteger(MediaFormat.KEY_FRAME_RATE, 30);
		f.setInteger(MediaFormat.KEY_I_FRAME_INTERVAL, 2);
		// Static screen still produces frames, so the player never starves.
		if (repeat) f.setLong(MediaFormat.KEY_REPEAT_PREVIOUS_FRAME_AFTER, 100_000);
		f.setInteger(MediaFormat.KEY_MAX_B_FRAMES, 0);
		f.setInteger(MediaFormat.KEY_PRIORITY, 0); // realtime
		return f;
	}
	void start() {
		codec.start();
		thread = new Thread(this::drain, "video-enc");
		thread.start();
	}
	void requestKeyFrame() {
		try {
		    Bundle b = new Bundle();
		    b.putInt(MediaCodec.PARAMETER_KEY_REQUEST_SYNC_FRAME, 0);
		    codec.setParameters(b);
		} catch (Exception ignored) {}
	}
	void stop() {
		stop = true;
		if (thread != null) {
		    try { thread.join(1000); } catch (InterruptedException ignored) {}
		}
		try { codec.stop(); } catch (Exception ignored) {}
		try { codec.release(); } catch (Exception ignored) {}
		surface.release();
	}
	private void drain() {
		MediaCodec.BufferInfo info = new MediaCodec.BufferInfo();
		byte[] sps = null, pps = null;
		boolean inited = false;
		long offsetUs = Long.MIN_VALUE;
		int seq = 1;
		long pendT = -1, lastEnd = 0, pendArrivalNs = 0; byte[] pendSample = null; boolean pendKey = false;
		try {
		    while (!stop) {
		        int idx = codec.dequeueOutputBuffer(info, 10_000);
		        if (idx == MediaCodec.INFO_OUTPUT_FORMAT_CHANGED) {
		            // Some encoders deliver SPS/PPS only as csd-0/csd-1 in the output format.
		            MediaFormat of = codec.getOutputFormat();
		            for (String k : new String[]{"csd-0", "csd-1"}) {
		                ByteBuffer c = of.containsKey(k) ? of.getByteBuffer(k) : null;
		                if (c == null) continue;
		                byte[] b = new byte[c.remaining()];
		                c.duplicate().get(b);
		                for (byte[] n : AnnexB.split(b, 0, b.length)) {
		                    if (AnnexB.type(n) == 7 && sps == null) sps = n;
		                    else if (AnnexB.type(n) == 8 && pps == null) pps = n;
		                }
		            }
		            if (!inited && sps != null && pps != null) {
		                sink.onVideoInit(Fmp4.videoInit(sps, pps, width, height), Fmp4.videoCodec(sps));
		                inited = true;
		            }
		            continue;
		        }
		        if (idx < 0) {
		            // Screen went still: don't hold the last frame back, publish it now.
		            if (pendSample != null && System.nanoTime() - pendArrivalNs > 50_000_000L) {
		                long nowT = (System.nanoTime() / 1000 - t0Us) * 9 / 100;
		                int dur = (int) Math.max(3000, Math.min(nowT - pendT, 90_000));
		                sink.onVideoFragment(Fmp4.fragment(seq++, pendT, dur, pendSample,
		                        pendKey ? Fmp4.FLAGS_SYNC : Fmp4.FLAGS_NON_SYNC), pendKey);
		                lastEnd = pendT + dur;
		                pendSample = null;
		            }
		            continue;
		        }
		        ByteBuffer bb = codec.getOutputBuffer(idx);
		        byte[] data = new byte[info.size];
		        if (bb != null && info.size > 0) {
		            bb.position(info.offset);
		            bb.limit(info.offset + info.size);
		            bb.get(data);
		        }
		        int flags = info.flags;
		        long pts = info.presentationTimeUs;
		        codec.releaseOutputBuffer(idx, false);
		        if ((flags & MediaCodec.BUFFER_FLAG_END_OF_STREAM) != 0) break;
		        if (data.length == 0) continue;
		        List<byte[]> nals = AnnexB.split(data, 0, data.length);
		        boolean key = (flags & MediaCodec.BUFFER_FLAG_KEY_FRAME) != 0;
		        for (byte[] n : nals) {
		            int t = AnnexB.type(n);
		            if (t == 7) sps = n;
		            else if (t == 8) pps = n;
		            else if (t == 5) key = true;
		        }
		        if (!inited && sps != null && pps != null) {
		            sink.onVideoInit(Fmp4.videoInit(sps, pps, width, height), Fmp4.videoCodec(sps));
		            inited = true;
		        }
		        if ((flags & MediaCodec.BUFFER_FLAG_CODEC_CONFIG) != 0 || !inited) continue;
		        byte[] sample = AnnexB.toAvcc(nals);
		        if (sample.length == 0) continue;
		        if (offsetUs == Long.MIN_VALUE) {
		            long now = System.nanoTime() / 1000;
		            offsetUs = Math.abs(now - pts) > 1_000_000 ? now - pts : 0;
		            if (offsetUs != 0) Log.w(TAG, "video pts not on monotonic clock, offset " + offsetUs);
		        }
		        long t = Math.max(0, (pts + offsetUs - t0Us) * 9 / 100); // us -> 90 kHz
		        if (pendSample == null && t <= lastEnd - 3000) t = lastEnd; // never go backwards after an idle flush
		        if (pendSample != null) {
		            if (t <= pendT) t = pendT + 1;
		            int dur = (int) Math.min(t - pendT, 90_000);
		            sink.onVideoFragment(Fmp4.fragment(seq++, pendT, dur, pendSample,
		                    pendKey ? Fmp4.FLAGS_SYNC : Fmp4.FLAGS_NON_SYNC), pendKey);
		        }
		        pendT = t; pendSample = sample; pendKey = key; pendArrivalNs = System.nanoTime();
		    }
		} catch (Exception e) {
		    if (!stop) Log.e(TAG, "video drain error", e);
		}
	}
}
